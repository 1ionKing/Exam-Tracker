// Map raw topics to object structure
let syllabusData = rawTopics.map((topic, index) => ({
  id: index + 1,
  subject: 'SSC CGL Syllabus', 
  topic: topic.trim(),
  completed: false
}));

// DOM Elements
const topicListEl = document.getElementById('topicList');
const progressCircleEl = document.getElementById('progressCircle');
const progressTextEl = document.getElementById('progressText');
const nextTopicNameEl = document.getElementById('nextTopicName');
const examDateInput = document.getElementById('examDateInput');
const daysLeftText = document.getElementById('daysLeftText');
const searchInput = document.getElementById('searchInput');

/**
 * Render Syllabus Topics with Optional Search Filter
 */
const renderTopics = (searchTerm = '') => {
  topicListEl.innerHTML = ''; 

  // Filter topics based on search input
  const filteredData = syllabusData.filter(item => 
    item.topic.toLowerCase().includes(searchTerm.toLowerCase())
  );

  if (filteredData.length === 0) {
    topicListEl.innerHTML = `<li class="topic-item"><span class="topic-item__subject">No topics found matching "${searchTerm}"</span></li>`;
    return;
  }

  filteredData.forEach(item => {
    const tagClass = item.completed ? 'topic-item__tag--completed' : 'topic-item__tag--pending';
    const tagText = item.completed ? 'Done' : 'Pending';
    const nameClass = item.completed ? 'is-completed' : '';
    const isChecked = item.completed ? 'checked' : '';

    const li = document.createElement('li');
    li.className = 'topic-item';
    
    li.innerHTML = `
      <div class="topic-item__left">
        <input type="checkbox" class="topic-item__checkbox" id="topic-${item.id}" ${isChecked}>
        <div class="topic-item__info">
          <span class="topic-item__subject">${item.subject}</span>
          <label class="topic-item__name ${nameClass}" for="topic-${item.id}">${item.topic}</label>
        </div>
      </div>
      <span class="topic-item__tag ${tagClass}">${tagText}</span>
    `;

    const checkbox = li.querySelector('.topic-item__checkbox');
    checkbox.addEventListener('change', (e) => toggleCompletion(item.id, e.target.checked));
    topicListEl.appendChild(li);
  });
};

const toggleCompletion = (id, isChecked) => {
  const topicIndex = syllabusData.findIndex(item => item.id === id);
  if (topicIndex !== -1) {
    syllabusData[topicIndex].completed = isChecked;
    // Re-render keeping the current search term active
    renderTopics(searchInput.value); 
    updateDashboard();
  }
};

/**
 * Calculate Progress Percentage
 */
const updateProgress = () => {
  if (syllabusData.length === 0) return;
  const totalTopics = syllabusData.length;
  const completedTopics = syllabusData.filter(item => item.completed).length;
  const percentage = Math.round((completedTopics / totalTopics) * 100);

  progressTextEl.textContent = `${percentage}%`;
  progressCircleEl.style.background = `conic-gradient(var(--accent-teal) ${percentage}%, var(--border-color) 0)`;
};

/**
 * Calculate Days Left based on user input
 */
const updateDaysLeft = () => {
  const selectedDate = new Date(examDateInput.value);
  const today = new Date();
  
  today.setHours(0, 0, 0, 0);
  selectedDate.setHours(0, 0, 0, 0);

  const diffTime = selectedDate - today;
  const diffDays = Math.ceil(diffTime / (1000 * 60 * 60 * 24));

  if (diffDays >= 0) {
    daysLeftText.textContent = diffDays;
    daysLeftText.style.color = "var(--accent-orange)";
  } else {
    daysLeftText.textContent = "0";
    daysLeftText.style.color = "#ef4444"; 
  }
};

/**
 * Update the Bottom Bar
 */
const updateBottomBar = () => {
  const nextTopic = syllabusData.find(item => !item.completed);
  if (nextTopic) {
    nextTopicNameEl.textContent = nextTopic.topic;
  } else {
    nextTopicNameEl.textContent = "All caught up!";
  }
};

const updateDashboard = () => {
  updateProgress();
  updateBottomBar();
};

// Event Listeners
examDateInput.addEventListener('change', updateDaysLeft);

// Search Bar Event Listener
searchInput.addEventListener('input', (e) => {
  renderTopics(e.target.value);
});

// Initialize
const initApp = () => {
  updateDaysLeft();
  renderTopics();
  updateDashboard();
};

document.addEventListener('DOMContentLoaded', initApp);
